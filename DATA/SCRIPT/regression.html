def read_table(path):
    if path.lower().endswith((".xlsx", ".xls")):
        return pd.read_excel(path)

    for enc in ["utf-8-sig", "utf-8", "cp1252", "latin1"]:
        try:
            return pd.read_csv(path, encoding=enc, low_memory=False)
        except Exception:
            pass

    raise ValueError(f"Cannot read file: {path}")


def clean_id(s):
    return (
        s.astype(str)
        .str.replace(".0", "", regex=False)
        .str.strip()
        .replace(["nan", "None", ""], np.nan)
    )


def to_num(s):
    return pd.to_numeric(
        s.astype(str)
        .str.replace(",", ".", regex=False)
        .str.strip(),
        errors="coerce"
    )


def find_col(df, candidates):
    lower = {c.lower(): c for c in df.columns}

    for cand in candidates:
        if cand.lower() in lower:
            return lower[cand.lower()]

    for col in df.columns:
        for cand in candidates:
            if cand.lower() in col.lower():
                return col

    return None


def haversine_m(lat1, lon1, lat2, lon2):
    R = 6371000

    lat1 = np.radians(lat1)
    lon1 = np.radians(lon1)
    lat2 = np.radians(lat2)
    lon2 = np.radians(lon2)

    dlat = lat2 - lat1
    dlon = lon2 - lon1

    a = (
        np.sin(dlat / 2) ** 2
        + np.cos(lat1) * np.cos(lat2) * np.sin(dlon / 2) ** 2
    )

    return 2 * R * np.arcsin(np.sqrt(a))


def fit_model(data, predictors, y_col="logY"):
    X = sm.add_constant(data[predictors])
    y = data[y_col]
    return sm.OLS(y, X).fit()


def rmse(model, data, predictors, y_col="logY"):
    pred = model.predict(sm.add_constant(data[predictors]))
    return np.sqrt(mean_squared_error(data[y_col], pred))


def max_vif(data, predictors):
    if len(predictors) <= 1:
        return 1.0

    try:
        X = data[predictors]
        return max([
            variance_inflation_factor(X.values, i)
            for i in range(X.shape[1])
        ])
    except Exception:
        return np.inf


def coef_table(model):
    return pd.DataFrame({
        "Variable": model.params.index,
        "Coefficient": model.params.values,
        "Std_Error": model.bse.values,
        "t_value": model.tvalues.values,
        "p_value": model.pvalues.values
    })


def build_equation(model):
    coef = model.params
    eq = f"log(1 + Y) = {coef['const']:.6f}"

    for var in coef.index:
        if var == "const":
            continue

        sign = "+" if coef[var] >= 0 else "-"
        eq += f" {sign} {abs(coef[var]):.6f} × {var}"

    demand_eq = "Y = exp(" + eq.split("=")[1].strip() + ") - 1"

    return eq, demand_eq

def stepwise_selection(
    data,
    candidate_vars,
    y_col="logY",
    threshold_in=0.05,
    threshold_out=0.10
):
    selected = []
    history = []

    while True:
        changed = False
        remaining = list(set(candidate_vars) - set(selected))

        pvals = pd.Series(index=remaining, dtype=float)

        for var in remaining:
            model = fit_model(data, selected + [var], y_col)
            pvals[var] = model.pvalues[var]

        if len(pvals) > 0:
            best_p = pvals.min()
            best_var = pvals.idxmin()

            if best_p < threshold_in:
                selected.append(best_var)
                changed = True

                history.append({
                    "Action": "ADD",
                    "Variable": best_var,
                    "p_value": best_p,
                    "Current_Model": ", ".join(selected)
                })

                print(f"ADD  {best_var:30s} p={best_p:.6f}")

        if len(selected) > 0:
            model = fit_model(data, selected, y_col)
            pvalues = model.pvalues.iloc[1:]

            worst_p = pvalues.max()
            worst_var = pvalues.idxmax()

            if worst_p > threshold_out:
                selected.remove(worst_var)
                changed = True

                history.append({
                    "Action": "DROP",
                    "Variable": worst_var,
                    "p_value": worst_p,
                    "Current_Model": ", ".join(selected)
                })

                print(f"DROP {worst_var:30s} p={worst_p:.6f}")

        if not changed:
            break

    return selected, pd.DataFrame(history)
